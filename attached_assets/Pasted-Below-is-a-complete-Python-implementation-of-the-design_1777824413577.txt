```python
"""
CSA S16 Steel Compression Member Design - Complete Workflow
Based on Chapter 3 (Compression Members) summary.
Author: (Your tool)
"""

import math

# =============================================================================
# CONSTANTS (CSA S16 typical values)
# =============================================================================
E = 200_000.0        # Young's modulus (MPa)
G = 77_000.0         # Shear modulus (MPa)
phi = 0.9            # resistance factor for compression

# =============================================================================
# 1. LOCAL BUCKLING: Width-to-thickness classification (Table 1 for axial)
# =============================================================================
def classify_section_axial(Fy, b_over_t_flange=None, b_over_t_web=None, b_over_t_angle=None,
                           D_over_t_circular=None, is_angle=False):
    """
    Determine section Class (1,2,3,4) for axial compression per CSA S16 Table 1.
    Returns Class number (4 if any limit exceeded).
    """
    # Limits from Table 1 (axial compression)
    limit_flange_I = 200 / math.sqrt(Fy)      # Flanges of I-sections, channels
    limit_web_I = 670 / math.sqrt(Fy)         # Web of I-shape (supported two edges)
    limit_angle_leg = 250 / math.sqrt(Fy)     # Legs of angles (supported one edge)
    limit_circular_HSS = 23000 / Fy           # D/t for circular HSS

    class_ = 1   # start as Class 1 (most compact); any exceedance drops to Class 4
    if b_over_t_flange is not None:
        if b_over_t_flange > limit_flange_I:
            class_ = 4
    if b_over_t_web is not None:
        if b_over_t_web > limit_web_I:
            class_ = 4
    if is_angle and b_over_t_angle is not None:
        if b_over_t_angle > limit_angle_leg:
            class_ = 4
    if D_over_t_circular is not None:
        if D_over_t_circular > limit_circular_HSS:
            class_ = 4
    return class_

# =============================================================================
# 2. EFFECTIVE LENGTH FACTOR (K) - Simple end conditions
# =============================================================================
def get_K(end_condition):
    """
    Return K factor based on standard end restraint.
    end_condition: 'pinned-pinned', 'fixed-pinned', 'fixed-fixed', 'fixed-free'
    For more advanced (frame) use user input directly.
    """
    mapping = {
        'pinned-pinned': 1.0,
        'fixed-pinned': 0.7,
        'fixed-fixed': 0.5,
        'fixed-free': 2.0,
    }
    return mapping.get(end_condition, 1.0)

# =============================================================================
# 3. FLEXURAL BUCKLING STRESS (Fe)
# =============================================================================
def flexural_buckling_stress(KL_r):
    """Euler buckling stress (MPa) given KL/r."""
    if KL_r <= 0:
        return float('inf')
    return (math.pi ** 2 * E) / (KL_r ** 2)

# =============================================================================
# 4. GENERAL RESISTANCE FORMULA (Cl. 13.3.1)
# =============================================================================
def compressive_resistance(A, Fy, Fe, n, phi=0.9):
    """
    Compute factored axial compressive resistance Cr (N).
    A : cross-sectional area (mm²)
    Fy: yield strength (MPa)
    Fe: elastic buckling stress (MPa)
    n  : parameter (1.34 or 2.24)
    """
    if Fe <= 0:
        return 0.0
    lam = math.sqrt(Fy / Fe)
    denominator = (1 + lam ** (2 * n)) ** (1.0 / n)
    Cr = phi * A * Fy / denominator
    return Cr

# =============================================================================
# 5. TORSIONAL AND FLEXURAL-TORSIONAL BUCKLING (Cl. 13.3.2)
# =============================================================================
def torsional_buckling_stress(Kz_Lz, A, r0_sq, Cw, J):
    """
    Fez for doubly symmetric sections (cruciform, Z).
    Kz_Lz = effective length for torsion (mm) = Kz * Lz
    r0_sq = polar radius of gyration squared (mm²) = x0² + y0² + rx² + ry²
    Cw = warping constant (mm⁶)
    J  = torsional constant (mm⁴)
    """
    if Kz_Lz == 0:
        return float('inf')
    term1 = (math.pi ** 2 * E * Cw) / (Kz_Lz ** 2)
    return (term1 + G * J) / (A * r0_sq)

def flexural_torsional_buckling_stress(Fey, Fez, x0, y0, r0_sq):
    """
    Feyz for singly symmetric sections (y axis of symmetry).
    Fey = weak-axis flexural buckling stress
    Fez = torsional buckling stress
    Returns Feyz (MPa)
    """
    Omega = 1 - (x0**2 + y0**2) / r0_sq
    if Omega <= 0:
        return min(Fey, Fez)
    # Formula from Clause 13.3.2(b)
    sum_ = Fey + Fez
    product = 4 * Fey * Fez * Omega
    radicand = 1 - product / (sum_ ** 2)
    if radicand < 0:
        radicand = 0
    Feyz = (sum_ / (2 * Omega)) * (1 - math.sqrt(radicand))
    return Feyz

# =============================================================================
# 6. SINGLE-ANGLE MEMBERS (Cl. 13.3.3) – Effective slenderness adjustment
# =============================================================================
def single_angle_KL_r_planar_truss(L, rx, b_long, b_short, connected_leg='longer'):
    """
    Compute effective KL/r for single angle in planar truss (Cl. 13.3.3.2).
    L   = member length between work points (mm)
    rx  = radius of gyration about geometric axis parallel to connected leg (mm)
    b_long, b_short = leg lengths (mm)
    connected_leg = 'longer' or 'shorter'
    Returns KL/r (unitless)
    """
    ratio = L / rx
    if connected_leg == 'longer':
        if ratio <= 80:
            KL_r = 72 + 0.75 * ratio
        else:
            KL_r = 32 + 1.25 * ratio
        KL_r = min(KL_r, 200)   # upper limit
    else:   # connected through shorter leg
        # first compute as if connected through longer leg
        if ratio <= 80:
            KL_r_base = 72 + 0.75 * ratio
        else:
            KL_r_base = 32 + 1.25 * ratio
        # increase factor
        leg_ratio = b_long / b_short
        increase = 4 * (leg_ratio**2 - 1)
        KL_r = KL_r_base + increase
        # lower bound
        # r_y' is radius of gyration about minor principal axis (user must provide)
        # Here we assume user will compare later; we return adjusted value.
        # For simplicity, we return KL_r and a flag.
    return KL_r

# =============================================================================
# MAIN DESIGN FUNCTION – puts everything together
# =============================================================================
def design_compression_member(
    # Basic properties
    A, Fy,                     # area (mm²), yield strength (MPa)
    Lx, Ly,                    # unsupported lengths (mm)
    Kx=1.0, Ky=1.0, Kz=1.0,   # effective length factors (user can override)
    rx=None, ry=None,          # radii of gyration (mm) – if None, they must be computed from I/A
    # Cross-section type and classification data
    section_type='doubly_symmetric',   # 'doubly_symmetric', 'singly_symmetric', 'angle'
    # For local buckling check (axial)
    b_over_t_flange=None, b_over_t_web=None, D_over_t=None,
    # For torsional buckling (required for non-doubly-symmetric)
    x0=0.0, y0=0.0,           # shear centre coordinates (mm)
    Cw=0.0, J=0.0,            # warping constant (mm⁶), torsional constant (mm⁴)
    # Single-angle specific (if section_type='angle')
    L_angle=None, rx_angle=None, b_long=None, b_short=None,
    connected_leg='longer', r_y_prime=None,   # r_y_prime for lower bound check
    # Parameters n (1.34 or 2.24)
    n_param=1.34,
):
    """
    Returns a dictionary with:
      - class_section (1 to 4)
      - Cr (factored compressive resistance in N)
      - Fe_used (lowest elastic buckling stress)
      - failure_mode ('yielding', 'flexural_buckling', 'torsional', 'flexural-torsional')
      - warnings (list)
    """
    warnings = []
    
    # -------- Step 1: Local buckling classification (axial) ----------
    if section_type == 'angle':
        # For angle, b/t is typically provided as b_over_t_angle. Here we use b_long/t.
        # We assume thickness t is same; user computes b/t.
        # We'll use b_over_t_flange as the leg b/t.
        class_ = classify_section_axial(Fy, b_over_t_flange=b_over_t_flange,
                                        is_angle=True, b_over_t_angle=b_over_t_flange)
    else:
        class_ = classify_section_axial(Fy, b_over_t_flange=b_over_t_flange,
                                        b_over_t_web=b_over_t_web,
                                        D_over_t_circular=D_over_t)
    
    if class_ == 4:
        warnings.append("Class 4 section – use CSA S16 Cl. 13.3.5 (not implemented here). Resistance may be overestimated.")
        # For simplicity we continue, but user should implement reduced effective width.
    
    # -------- Step 2: Compute slenderness ratios ----------
    if rx is None or ry is None:
        raise ValueError("rx and ry must be provided if radii of gyration not given")
    KLx_rx = (Kx * Lx) / rx if rx > 0 else float('inf')
    KLy_ry = (Ky * Ly) / ry if ry > 0 else float('inf')
    
    # -------- Step 3: Determine critical elastic buckling stress Fe ----------
    Fe_ex = flexural_buckling_stress(KLx_rx)   # strong axis
    Fe_ey = flexural_buckling_stress(KLy_ry)   # weak axis
    
    Fe_ez = None
    Fe_eyz = None
    failure_mode = 'flexural_buckling'
    
    if section_type == 'doubly_symmetric':
        # Only flexural buckling (Cl. 13.3.1). But also check if torsional may govern (e.g. cruciform)
        # However for typical doubly symmetric I/HSS, flexural governs. We'll take min(Fe_ex, Fe_ey)
        Fe = min(Fe_ex, Fe_ey)
        if Fe == Fe_ex:
            failure_mode = 'flexural_buckling_strong_axis'
        else:
            failure_mode = 'flexural_buckling_weak_axis'
    
    elif section_type == 'singly_symmetric':
        # Need to compute Fez and Feyz (Cl. 13.3.2(b))
        # Assume y-axis is axis of symmetry
        # Compute r0^2 (polar radius of gyration squared)
        if rx and ry:
            r0_sq = x0**2 + y0**2 + rx**2 + ry**2
        else:
            r0_sq = 0  # fallback
        if r0_sq <= 0:
            warnings.append("Invalid r0^2 for torsional buckling, using flexural only")
            Fe = min(Fe_ex, Fe_ey)
        else:
            Fez = torsional_buckling_stress(Kz*max(Lx, Ly), A, r0_sq, Cw, J)
            Fe_eyz = flexural_torsional_buckling_stress(Fe_ey, Fez, x0, y0, r0_sq)
            Fe = min(Fe_ex, Fe_eyz)
            if Fe == Fe_eyz:
                failure_mode = 'flexural_torsional_buckling'
            else:
                failure_mode = 'flexural_buckling_strong_axis'
    
    elif section_type == 'angle':
        # Single-angle members (Cl. 13.3.3) – use effective KL/r from special rules
        if L_angle is None or rx_angle is None or b_long is None or b_short is None:
            warnings.append("Missing data for single-angle effective length, using ordinary flexural buckling")
            Fe = min(Fe_ex, Fe_ey)
        else:
            # Compute effective KL/r using planar truss formula (Cl. 13.3.3.2)
            KL_r_eff = single_angle_KL_r_planar_truss(L_angle, rx_angle, b_long, b_short, connected_leg)
            # Lower bound check: must not be less than 0.95*L/r_y'
            if r_y_prime is not None and L_angle is not None:
                lower_bound = 0.95 * L_angle / r_y_prime
                if KL_r_eff < lower_bound:
                    KL_r_eff = lower_bound
                    warnings.append("KL/r increased to lower bound 0.95*L/r_y'")
            # Compute Fe from this effective KL/r
            Fe = flexural_buckling_stress(KL_r_eff)
            failure_mode = 'flexural_buckling_angle_effective'
    
    else:
        # Asymmetric sections – simplified: use min of Fe_ex, Fe_ey (full cubic not implemented)
        warnings.append("Asymmetric section: full cubic equation not implemented – using min flexural Fe")
        Fe = min(Fe_ex, Fe_ey)
    
    # -------- Step 4: Compute compressive resistance Cr ----------
    Cr = compressive_resistance(A, Fy, Fe, n_param, phi)
    
    # Optional: cap by pure yield (though formula already does)
    Cr_yield = phi * A * Fy
    final_Cr = min(Cr, Cr_yield)
    
    return {
        'class_section': class_,
        'Cr (N)': final_Cr,
        'Fe_used (MPa)': Fe,
        'failure_mode': failure_mode,
        'warnings': warnings,
        'slenderness_ratios': {'KLx/rx': KLx_rx, 'KLy/ry': KLy_ry}
    }

# =============================================================================
# EXAMPLE USAGE (W250x73, Grade 350W, 8m pinned-pinned)
# =============================================================================
if __name__ == "__main__":
    # Example 1 from chapter (W250x73, L=8m, pinned ends)
    # Properties for W250x73 (approx – user should use actual from handbook)
    A = 9280          # mm²
    Fy = 350          # MPa (Grade 350W)
    rx = 110          # mm (approx)
    ry = 64.5         # mm (approx)
    L = 8000          # mm
    K = get_K('pinned-pinned')
    
    # b/t for flange and web (example values)
    b_over_t_flange = 8.5      # typical
    b_over_t_web = 35.0        # typical
    
    result = design_compression_member(
        A=A, Fy=Fy,
        Lx=L, Ly=L,
        Kx=K, Ky=K,
        rx=rx, ry=ry,
        section_type='doubly_symmetric',
        b_over_t_flange=b_over_t_flange,
        b_over_t_web=b_over_t_web,
        n_param=1.34      # hot-rolled
    )
    
    print("=== CSA S16 Compression Member Design ===")
    for k, v in result.items():
        print(f"{k}: {v}")
    print(f"Factored compressive resistance Cr = {result['Cr (N)']/1000:.1f} kN")
```
