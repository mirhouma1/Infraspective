function omega2General(Mmax, Ma, Mb, Mc) {
const d = Math.sqrt(Mmax**2 + 4*Ma**2 + 7*Mb**2 + 4*Mc**2);
return Math.min(4 * Mmax / d, 2.5);
}
function omega2Linear(k) {
return Math.min(1.75 + 1.05*k + 0.3*k*k, 2.5);
}
function criticalMoment(L, Iy, J, Cw, w2) {
const term1 = E * Iy * G * J;
const term2 = (Math.PI * E / L)**2 * Iy * Cw;
return (w2 * Math.PI / L) * Math.sqrt(term1 + term2) / 1e6;
}

// Shear Fs vs h/w diagram
function ShearFsDiagram({ Fy, stiffened, kv }) {
const W = 300, H = 160;
const pad = { l: 45, r: 15, t: 15, b: 35 };
const iW = W - pad.l - pad.r, iH = H - pad.t - pad.b;

const sqFy = Math.sqrt(Fy);
const maxRatio = 200;
const steps = 120;
const points = [];

for (let i = 0; i <= steps; i++) {
const r = (i / steps) * maxRatio;
if (r === 0) continue;
let Fs;
if (!stiffened) {
if (r <= 1014 / sqFy) Fs = 0.66 * Fy;
else if (r <= 1435 / sqFy) Fs = 670 * sqFy / r;
else Fs = 961200 / (r * r);
} else {
const kvv = kv || 5.34;
const Fcri = 290 * Math.sqrt(Fy * kvv) / r;
const Fcre = 180000 * kvv / (r * r);
const l1 = 439 * Math.sqrt(kvv / Fy);
const l2 = 502 * Math.sqrt(kvv / Fy);
const l3 = 621 * Math.sqrt(kvv / Fy);
if (r <= l1) Fs = 0.66 * Fy;
else if (r <= l2) Fs = Fcri;
else if (r <= l3) Fs = Fcri + (1 / Math.sqrt(1 + (r/100)**2)) * (0.5*Fy - 0.866*Fcri);
else Fs = Fcre + (1 / Math.sqrt(1 + (r/100)**2)) * (0.5*Fy - 0.866*Fcre);
}
Fs = Math.max(0, Math.min(Fs, 0.66 * Fy));
points.push({ r, Fs });
}

const maxFs = 0.66 * Fy;
const sx = r => pad.l + (r / maxRatio) * iW;
const sy = f => pad.t + iH - (f / maxFs) * iH;

const pts = points.map(p => `${sx(p.r).toFixed(1)},${sy(p.Fs).toFixed(1)}`).join(” “);

return (
<svg width={W} height={H} style={{ display: “block”, margin: “0 auto” }}>
{[0.25, 0.5, 0.75, 1].map(f => (
<line key={f} x1={pad.l} y1={sy(maxFs * f)} x2={W - pad.r} y2={sy(maxFs * f)}
stroke={C.border} strokeWidth={0.5} />
))}
<polyline points={pts} fill="none" stroke={C.accent} strokeWidth={2} />
<line x1={pad.l} y1={pad.t} x2={pad.l} y2={H - pad.b} stroke={C.muted} strokeWidth={1} />
<line x1={pad.l} y1={H - pad.b} x2={W - pad.r} y2={H - pad.b} stroke={C.muted} strokeWidth={1} />
<text x={W / 2} y={H - 4} textAnchor=“middle” fill={C.muted} fontSize={10}>h/w ratio</text>
<text x={10} y={H / 2} textAnchor=“middle” fill={C.muted} fontSize={10}
transform={`rotate(-90, 10, ${H/2})`}>Fs (MPa)</text>
<text x={pad.l + 4} y={sy(maxFs) - 3} fill={C.green} fontSize={9}>0.66Fy = {(0.66*Fy).toFixed(0)}</text>
</svg>
);
}

function ShearPanel() {
const [h, setH] = useState(400);
const [w, setW] = useState(8);
const [Fy, setFy] = useState(350);
const [stiffened, setStiffened] = useState(false);
const [a, setA] = useState(600);

const sqFy = Math.sqrt(Fy);
const ratio = h / w;

let Fs, caseLabel, Fcri, Fcre, kv;
if (!stiffened) {
const l1 = 1014 / sqFy, l2 = 1435 / sqFy;
if (ratio <= l1) { Fs = 0.66 * Fy; caseLabel = “Case i: Fs = 0.66Fy”; }
else if (ratio <= l2) { Fs = 670 * sqFy / ratio; caseLabel = “Case ii: Fs = 670√Fy / (h/w)”; }
else { Fs = 961200 / (ratio**2); caseLabel = “Case iii: Fs = 961200 / (h/w)²”; }
} else {
const aspect = a / h;
kv = aspect < 1 ? 4 + 5.34 / aspect**2 : 5.34 + 4 / aspect**2;
const ka = 1 / Math.sqrt(1 + aspect**2);
Fcri = 290 * Math.sqrt(Fy * kv) / ratio;
Fcre = 180000 * kv / (ratio**2);
const l1 = 439 * Math.sqrt(kv / Fy), l2 = 502 * Math.sqrt(kv / Fy), l3 = 621 * Math.sqrt(kv / Fy);
if (ratio <= l1) { Fs = 0.66 * Fy; caseLabel = “Case i: Fs = 0.66Fy”; }
else if (ratio <= l2) { Fs = Fcri; caseLabel = “Case ii: Fs = Fcri”; }
else if (ratio <= l3) { Fs = Fcri + ka*(0.5*Fy - 0.866*Fcri); caseLabel = “Case iii: Post-buckling”; }
else { Fs = Fcre + ka*(0.5*Fy - 0.866*Fcre); caseLabel = “Case iv: Elastic buckling”; }
}

const Aw = h * w;
const Vr = PHI * Aw * Fs / 1000;

return (
<div>
<div style={{ display: “grid”, gridTemplateColumns: “1fr 1fr”, gap: 12, marginBottom: 16 }}>
<div>
<div style={{ marginBottom: 12 }}>

background: (stiffened === (t === “Stiffened”)) ? C.accent : “transparent”,
border: `1px solid ${C.border}`, color: (stiffened === (t === “Stiffened”)) ? “#fff” : C.muted,
borderRadius: 5, padding: “5px 12px”, fontSize: 12, cursor: “pointer”
}}>{t}</button>
))}
</div>
</div>
<Input label="Web depth h" value={h} onChange={setH} unit="mm" min={100} max={2000} step={10} />
<Input label="Web thickness w" value={w} onChange={setW} unit="mm" min={4} max={30} />
<Input label="Yield strength Fy" value={Fy} onChange={setFy} unit="MPa" min={250} max={690} step={50} />
{stiffened && <Input label="Stiffener spacing a" value={a} onChange={setA} unit="mm" min={100} max={3000} step={50} />}
</div>

```
    <div style={{ display: "flex", flexDirection: "column", gap: 10 }}>
      <Card>
        <Label>h / w</Label>
        <Value color={C.text}>{ratio.toFixed(1)}</Value>
      </Card>
      <Card>
        <Label>Shear yield stress Fs</Label>
        <Value color={C.accent}>{Fs.toFixed(1)} <span style={{ fontSize: 14, fontWeight: 400 }}>MPa</span></Value>
      </Card>
      <Card>
        <Label>Factored shear resistance Vr</Label>
        <Value color={C.green}>{Vr.toFixed(1)} <span style={{ fontSize: 14, fontWeight: 400 }}>kN</span></Value>
      </Card>
      <div style={{ padding: "8px 12px", background: "#1c2128", borderRadius: 6, borderLeft: `3px solid ${C.yellow}` }}>
        <div style={{ fontSize: 11, color: C.yellow }}>{caseLabel}</div>

    </div>
  </div>

  <Card style={{ marginTop: 4 }}>
```

);
}

function MomentPanel() {
const [sectionClass, setSectionClass] = useState(1);
const [Z, setZ] = useState(1200e3);
const [S, setS] = useState(1050e3);
const [Fy, setFy] = useState(350);
const [Iy, setIy] = useState(12e6);
const [J, setJ] = useState(600e3);
const [Cw, setCw] = useState(800e9);
const [L, setL] = useState(6000);
const [lateralMode, setLateralMode] = useState(“supported”);
const [k, setK] = useState(1.0);
const [loadType, setLoadType] = useState(“udl”);

const Mp = Z * Fy / 1e6;
const My = S * Fy / 1e6;
const w2 = omega2Linear(k);

let Mr, Mu_val, caseStr;
if (lateralMode === “supported”) {
if (sectionClass <= 2) { Mr = PHI * Mp; caseStr = “Cl. 13.5a: Mr = φMp”; }
else { Mr = PHI * My; caseStr = “Cl. 13.5b: Mr = φMy”; }
Mu_val = null;
} else {
Mu_val = criticalMoment(L, Iy, J, Cw, w2);
const Mref = sectionClass <= 2 ? Mp : My;
const thresh = 0.67 * Mref;
if (Mu_val > thresh) {
Mr = Math.min(1.15 * PHI * Mref * (1 - 0.28 * Mref / Mu_val), PHI * Mref);
caseStr = `Cl. 13.6: Mu > 0.67M${sectionClass<=2?"p":"y"} → inelastic LTB`;
} else {
Mr = PHI * Mu_val;
caseStr = `Cl. 13.6: Mu ≤ 0.67M${sectionClass<=2?"p":"y"} → elastic LTB`;
}
}

// Build Mr vs L curve
const Mr_vals = [];
for (let Li = 500; Li <= 15000; Li += 200) {
const Mu_i = criticalMoment(Li, Iy, J, Cw, w2);
const Mref = sectionClass <= 2 ? Mp : My;
let Mr_i;
if (Mu_i > 0.67 * Mref) {
Mr_i = Math.min(1.15 * PHI * Mref * (1 - 0.28 * Mref / Mu_i), PHI * Mref);
} else {
Mr_i = PHI * Mu_i;
}
Mr_vals.push({ L: Li, Mr: Mr_i });
}

return (

borderRadius: 5, padding: “5px 12px”, fontSize: 12, cursor: “pointer”, fontWeight: 700
}}>{c}</button>
))}
</div>
</div>

<Label>Support condition</Label>

{[[“supported”,“Laterally Supported”],[“unsupported”,“Laterally Unsupported”]].map(([v,l]) => (
<button key={v} onClick={() => setLateralMode(v)} style={{
background: lateralMode === v ? C.accent : “transparent”,
border: `1px solid ${lateralMode === v ? C.accent : C.border}`,
color: lateralMode === v ? “#fff” : C.muted,
borderRadius: 5, padding: “5px 12px”, fontSize: 12, cursor: “pointer”
}}>{l}</button>
))}
</div>
</div>
</div>
